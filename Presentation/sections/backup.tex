\begin{frame}{Motivazioni e limiti dell'approccio degree-day}

  \begin{columns}[c, onlytextwidth]

    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[height=0.7\textheight]{images/backup/energy_balance.jpeg}

      \vspace{-4pt}
      \[\frac{d\mathcal{H}}{dt}\,=\,R_N\,+\,H_S\,+\,H_L\,+\,H_P\,+\,G\]
    \end{column}

    \begin{column}{0.42\textwidth}
      \small
      \begin{block}{Perché $T_\text{aria}$}
        \begin{itemize}[label=$\blacktriangleright$, itemsep=3pt]
          \item misurata di routine
          \item proxy di $\text{LW}_\text{in} \propto \varepsilon\sigma T^4$
          \item DDF variabile per $\text{SW}_\text{in}$
        \end{itemize}
      \end{block}

      \vspace{4pt}
      \begin{block}{Limiti}
        \begin{itemize}[label=$\blacktriangleright$, itemsep=3pt]
          \item $\text{SW}_\text{in}$ e albedo non espliciti
          \item ignorati vento, umidità ed esposizione
          \item affidabile su scala giornaliera o stagionale, meno su scala oraria
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE ASSUNZIONI SIMULATED ANNEALING
\begin{frame}{Simulated annealing: assunzioni}
  
  {\usebeamercolor[fg]{structure}\large\bfseries Spazio di ricerca: \par}
  \begin{itemize}[label=$\blacktriangleright$, itemsep=3pt]
    \item Parametri:    $T_\tau$, $\text{DDF}_{ave}$, $\text{DDF}_{amp}$
    \item Valori consentiti:    $T_\tau \in (-4,\,8)$, \; $\text{DDF}_{ave} \in (1,\,9)$, \; $\text{DDF}_{amp} \in (0.5,\,7.5)$
    \item Proposta mossa:    $x'_i \sim \mathcal{U}(x_i - \delta_i,\, x_i + \delta_i)$, \quad $\delta_{T_\tau} = 2\,\delta_{\text{DDF}}$
  \end{itemize}

  \bigskip
  \medskip
  {\usebeamercolor[fg]{structure}\large\bfseries Schema di raffreddamento: \par}
    \begin{itemize}[label=$\blacktriangleright$, itemsep=3pt]
    \item Temperatura:    $T_k = 0.95^k\, T_0$, \quad $T_0 = 5$
    \item Cambio temperatura:    dopo 5 mosse accettate (max 500 tentativi)
    \item Delta adattivo:    $\delta \leftarrow \delta\, e^{\,\mathrm{AR} - 0.4}$, \quad $\delta \in [10^{-5},\, 1]$
    \item Arresto: $T_k < 5 \cdot 10^{-6}$ \; $\Rightarrow$ \; 270 livelli, 1350 mosse accettate
  \end{itemize}

\end{frame}
