% SLIDE SU LOCALLY WEIGHTED LINEAR REGRESSION
\begin{frame}[t]{Local Weighted Linear Regression}

  \vspace{-4pt}
  \begin{center}
    {\small Regressione locale precipitazione--quota
      $\;\longrightarrow\;$ climatologie mensili 1991--2020}\\[4pt]
    $P_{\text{clima}}(\mathbf{x}) \;=\; q(\mathbf{x}) \;+\; m(\mathbf{x})\, h(\mathbf{x})$
  \end{center}

  \vspace{4pt}
  \begin{columns}[c, onlytextwidth]

    \begin{column}{0.48\textwidth}
       \begin{block}{Pesi delle stazioni}
        \vspace{-4pt}
        {\footnotesize
        \[
          w_i = \prod_{par} w_i^{par}, \qquad
          w_i^{par} = \exp\left[-\frac{\left(\Delta_i^{par}\right)^2}{c^{par}}\right]
        \]}
        \vspace{-6pt}
        \begin{itemize}[label=$\blacktriangleright$, itemsep=0.3em]
          \raggedright
          \item {\small 5 fattori: distanza, quota, pendenza, esposizione, distanza dal mare}
          \item {\small $c^{par}$ ottimizzati localmente}
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[
          baseline=(current bounding box.center),
          font=\small, x=1cm, y=0.78cm,   % y=… invece di yscale: i cerchi restano cerchi
          ax/.style={-{Stealth[length=5pt,width=4pt]}, line width=0.8pt, black!70},
          lbl/.style={font=\footnotesize, text=black!75},
        ]
        \usebeamercolor{structure}\colorlet{acc}{structure.fg}
        \draw[ax] (0,0) -- (5.4,0) node[lbl, below left] {Quota};
        \draw[ax] (0,0) -- (0,3.6) node[lbl, above right] {$P_{\text{clima}}$};

        \foreach \x/\y/\w in {0.5/0.80/0.20, 0.9/1.35/0.30, 1.4/1.20/0.50,

                              2.9/1.90/0.95, 3.2/2.30/0.80, 3.5/2.05/0.60,
                              3.9/2.60/0.40, 4.4/2.55/0.25, 4.8/3.00/0.15,
                              1.2/2.20/0.15, 3.7/1.35/0.15}{
          \pgfmathsetmacro{\op}{0.25+0.75*\w}
          \fill[acc, fill opacity=\op] (\x,\y) circle ({1.2pt+3pt*\w});
