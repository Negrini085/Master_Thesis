\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.85\linewidth]{snow_products/sod_sed_by_ele_MODIS.png}
    \caption{Quantiles of the average Snow Onset Date (SOD, panel a) and Snow End Date (SED, panel b) distributions within each 
    $\SI{5}{\meter}$ wide elevation band over the 2001-2025 period. Both panels show a clear monotonic relationship with elevation, 
    since high altitude alpine catchments experience earlier onsets and later ends of the snow season. The relatively large spread 
    within each band indicates that elevation alone cannot fully account for the observed variability in snow cover, highlighting 
    the need to consider additional topographic variables.}
    \label{fig:sod_sed_by_ele_MODIS}
\end{figure}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\linewidth]{snow_products/diff_from_ele_sod_sed_MODIS.png}
    \caption{(a) Spatial distribution of the difference between the SOD value of each individual pixel and the mean SOD of the 
    corresponding $\SI{5}{\meter}$ wide elevation band. (b) Spatial distribution of the difference between the SED value of each 
    individual pixel and the mean SED of the corresponding $\SI{5}{\meter}$ wide elevation band. The Eastern Apennines typically 
    experience an earlier onset and a later end of the snow season relative to the elevation wise Italian average. This behaviour 
    could be relater to cold air masses originating from the Balkans. In the Alps, pixel aspect plays a significant role, as south 
    facing slopes often experience earlier melt.}
    \label{fig:diff_from_ele_sod_sed_MODIS}
\end{figure}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\linewidth]{snow_products/sd_sod_sed_with_ele.png}
    \caption{(a - b) Spatial distribution and quantiles elevation wise distribution of the standard deviation of SOD from the mean, computed over the 
    2001-2025 period. (c-d) Spatial distribution and quantiles elevation wise distribution of the standard deviation of SED from the mean, computed 
    over the 2001-2025 period. Both metrics exhibit an almost monotonic increase up to $\SI{1800}{\meter}$ a.s.l., beyond which a substantial 
    decrease can be detected. It's worth noting that the SOD decrease is monotonic, whereas SED exhibits a local maximum at $\SI{3000}{\meter}$. This 
    secondary peak could be related to ice-covered pixels that are now ice-free as a consequence of retreating glaciers in a warming enviroment, thereby 
    experiencing ordinary seasonal snow cover rather than a year-round one.}
    \label{fig:sd_sod_sed_with_ele}
\end{figure}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\linewidth]{snow_products/trends_sod_MODIS.png}
    \caption{(a) SOD trend slope computed by means of the non-parameter Sen-Theil estimator over the 2001--2025 period. (b) Trend sign and significance 
    over the 2001-2025 period, computed by means of non-parametric Mann-Kendall test. SOD values are increasing over the Apennines and the Western Alps, 
    meaning that fewer snow cover days occur between the 1 October and 25 January, whereas in the Eastern Alps there is no clear overall trend, as 
    increasing SOD values are counterbalanced by decreasing ones.}
    \label{fig:trends_sod_MODIS}
\end{figure}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\linewidth]{snow_products/trends_sed_MODIS.png}
    \caption{(a) SED trend slope computed by means of the non-parameter Sen-Theil estimator over the 2001--2025 period. (b) Trend sign and significance 
    over the 2001-2025 period, computed by means of non-parametric Mann-Kendall test. SED values are decreasing over the whole Italian Peninsula, with 
    just a few exceptions in the north-east. By comparing these maps with those of SED and SCD values, it becomes evident that the main contributor to 
    the reduction in snow season length is an earlier end of the snow season, rather than a delayed onset of snow cover.}
    \label{fig:trends_sed_MODIS}
\end{figure}
